import tkinter as tk from tkinter import filedialog import subprocess import sys def choose_file(): file_path = filedialog.askopenfilename( title="اختر ملف Python", filetypes=[("Python files", "*.py")] ) if not file_path: return output.delete("1.0", tk.END) output.insert(tk.END, f"تشغيل: {file_path}\n\n") try: result = subprocess.run( [sys.executable, file_path], capture_output=True, text=True, timeout=30 ) output.insert(tk.END, result.stdout) if result.stderr: output.insert(tk.END, "\n--- ERROR ---\n") output.insert(tk.END, result.stderr) except subprocess.TimeoutExpired: output.insert(tk.END, "\n❌ توقف: الملف تجاوز 30 ثانية.") except Exception as e: output.insert(tk.END, f"\n❌ خطأ: {e}") root = tk.Tk() root.title("Python File Runner") root.geometry("700x500") button = tk.Button( root, text="📂 اختر ملف Python وشغّله", command=choose_file, font=("Arial", 14) ) button.pack(pady=20) output = tk.Text(root, font=("Courier", 11)) output.pack(fill="both", expand=True, padx=15, pady=10) root.mainloop()
